\section{Problem \& Architecture}

%---------------------------------------------------------
\begin{frame}
\frametitle{Task \& Scoring Rule}

\begin{columns}[T]
\column{0.45\textwidth}
\textbf{Input:} \texttt{task.json} + heterogeneous context\\
{\small (CSV, SQLite, JSON, Markdown rules, record-text prose, images)}

\vspace{0.3em}
\textbf{Output:} \texttt{prediction.csv} scored by\\
\textbf{column-content signature matching}

\vspace{0.3em}
\begin{itemize}
    \item Column names \textbf{ignored}; row order \textbf{ignored}
    \item Multi-column match via bipartite augmenting paths
    \item Normalize: numeric $\to 10^{-2}$, string \texttt{strip+lower}
\end{itemize}

\column{0.52\textwidth}
\vspace{0.5em}
\begin{equation*}
\text{score} = \max\!\Big(0,\; \text{recall} - \lambda \cdot \tfrac{\text{extra\_cols}}{\text{pred\_cols}}\Big), \;\; \lambda = 0.5
\end{equation*}

\vspace{0.3em}
\begin{alertblock}{Design implications}
\begin{itemize}
    \item \textbf{Fewer columns} --- every extra wrong col is penalized
    \item \textbf{Preserve granularity} --- \texttt{first\_name}+\texttt{last\_name} beats merged \texttt{name}
    \item \textbf{Match dtypes} --- \texttt{163109.0} $\neq$ \texttt{163109}
\end{itemize}
\end{alertblock}
\end{columns}

\end{frame}
%---------------------------------------------------------

\begin{frame}
\frametitle{Why Layered? --- Four-Layer Architecture}

\textbf{Vanilla ReAct breaks down:} easy tasks waste tokens on multi-turn tool calls; hard tasks drift in intermediate steps; no structural guarantee on output shape; no cheap way to verify semantic correctness.

\vspace{0.4em}
\begin{center}

    arrow/.style={-{Stealth[length=5pt]}, thick, ustblue},
]
\node[box, fill=ustblue!15] (compiler) {1. TaskCompiler\\{\scriptsize deterministic profiling}};
\node[box, fill=ustblue!15, right=of compiler] (router) {2. Router\\{\scriptsize shape-based dispatch}};
\node[box, fill=ustgold_1!20, right=of router] (operator) {3. Operator Pipeline\\{\scriptsize codegen + repair}};
\node[box, fill=ustred_1!15, right=of operator] (guard) {4. Risk Gating\\{\scriptsize guard + judge}};

\end{center}

\begin{block}{Design principle}
\emph{Simple tasks pay minimal LLM cost; only suspicious answers escalate to expensive semantic verification.}
\end{block}

\section{Pipeline \& Routing}

%---------------------------------------------------------
\begin{frame}
\frametitle{TaskCompiler \& Router --- Dispatch by Shape, Not Difficulty}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{TaskCompiler (zero LLM):}
\begin{itemize}
    \item Per-file scan $\to$ \texttt{SourceCapability}\\{\scriptsize headers, dtypes, cardinality, samples}
    \item Task-level aggregation $\to$ \texttt{ExecutionProfile}
    \begin{itemize}
        \scriptsize
        \item \texttt{context\_size}, \texttt{source\_shape}
        \item \texttt{verifiability}, \texttt{operation\_complexity}
        \item \texttt{semantic\_rule\_required}
    \end{itemize}
    \item \alert{Single source of truth} --- reused by prompt, static checker, guard, judge
\end{itemize}

\column{0.48\textwidth}
\textbf{Router rules (hard, deterministic):}
\begin{itemize}
    \item \texttt{weak} / \texttt{image} $\to$ multi\_agent
    \item \texttt{long\_docs} $\to$ RAG-enabled operator
    \item \texttt{semantic\_rule\_required} $\to$ tool\_first\_mixed
    \item \texttt{small+programmatic+low} $\to$ easy
\end{itemize}

\vspace{0.3em}
\textbf{Cascade by \emph{failure type}, not difficulty:}\\
{\scriptsize zero\_rows $\to$ \alert{no upgrade} (column wrong, not route)\\
retrieval\_empty $\to$ RAG route\\
semantic\_consistency\_failed $\to$ tool\_first\_mixed\\
syntax/static/exec error $\to$ sibling operator}
\end{columns}

\vspace{0.3em}
\begin{exampleblock}{Cross-model verify (optional)}
Primary + verifiers run in parallel; keep columns whose signatures agree $\geq$ \texttt{min\_agreement} --- directly targets the $-\lambda \cdot$ extra penalty.
\end{exampleblock}

\end{frame}

\end{tikzpicture}
\end{center}

\vspace{0.2em}
\begin{block}{Shared evidence --- \texttt{debug\_steps}}
Every generated program emits \texttt{schema\_inspection}, \texttt{used\_columns}, \texttt{filter\_conditions}, \texttt{join\_keys}, \texttt{intermediate\_counts}. Read by \alert{3 consumers}: cheap guard, preflight, LLM judge --- single audit trail, no drift.
\end{block}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}
\frametitle{Schema Grounding + Static Checker + Local Repair}

\begin{columns}[T]
\column{0.32\textwidth}
\textbf{Schema Grounding}
\begin{itemize}
    \item Extract question concepts
    \item Match to real columns: Levenshtein + value match + cardinality
    \item Top-3 candidates w/ samples
    \item \alert{Fights hallucination}
\end{itemize}

\column{0.32\textwidth}
\textbf{Static Checker}
\begin{itemize}
    \item Python AST + SQL scan
    \item \texttt{no\_such\_column}, \texttt{bad\_join\_key}, \texttt{\_x/\_y} suffixes
    \item Runs \emph{before} execution
    \item Blocks predictable failures
\end{itemize}

\column{0.32\textwidth}
\textbf{Local Repair (0 LLM)}
\begin{itemize}
    \item 10+ deterministic patches
    \item Fuzzy column-name fix
    \item Cast merge keys to string
    \item Append missing \texttt{answer}
    \item Zero-row filter normalize
\end{itemize}
\end{columns}

\vspace{0.5em}
\begin{alertblock}{Why these three work together}
All three read the \textbf{same} \texttt{SourceCapability} scan. The prompt tells the LLM what's real; the static checker verifies the LLM didn't invent; local repair fixes the remaining deterministic mistakes without an extra LLM call.
\end{alertblock}

\end{frame}

\section{Risk Gating (Core Innovation)}

%---------------------------------------------------------
\begin{frame}
\frametitle{Cheap Semantic Guard --- Non-LLM Risk Gate}

\textbf{Position:} codegen succeeded + answer structurally valid + \alert{no analyst ran yet} $\to$ should we trust this answer?

\vspace{0.2em}
\begin{equation*}
\text{risk\_score} = \min\!\big(1.0,\textstyle\sum \text{unique\_weights}\big), \quad
\text{escalate} = \text{any error} \,\lor\, \text{score} \geq 0.5
\end{equation*}

\vspace{0.2em}
\begin{center}
\scriptsize
\begin{tabular}{llcl}
\toprule
\textbf{Category} & \textbf{Risk Code} & \textbf{Wt} & \textbf{Meaning} \\
\midrule
Execution & \texttt{execution\_failed / empty\_answer\_rows} & 1.0 / 0.75 & No/zero answer rows \\
Repair & \texttt{suspicious\_fallback:schema\_retry} & 0.8 & Needed schema retry to succeed \\
Trace & \texttt{trace\_missing:schema\_inspection} & 0.8 & No real-column check in trace \\
Trace & \texttt{zero\_intermediate\_count} & 0.8 & Mid-pipeline drops to 0 rows \\
\textbf{Coverage} & \texttt{\textbf{field\_coverage\_mismatch}} & \textbf{0.75} & \textbf{Grounding found candidate but code didn't use it} \\
Formula & \texttt{formula\_without\_trigger} & 0.75 & Used formula question didn't ask for \\
\bottomrule
\end{tabular}
\end{center}

\begin{alertblock}{Killer example}
Q: ``patients with \textbf{severe degree of thrombosis}'' --- code uses \texttt{disease == "thrombosis"} but grounding found \texttt{degree\_of\_thrombosis} as candidate $\to$ \texttt{field\_coverage\_mismatch} (0.75) $\to$ \alert{escalate to judge} instead of silently shipping wrong answer.
\end{alertblock}

\end{frame}

\begin{frame}
\frametitle{Lazy Escalation + Analyst-Exception Recovery}

\begin{columns}[T]
\column{0.55\textwidth}
\begin{center}
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[
    node distance=0.45cm and 1.0cm,
    box/.style={rectangle, draw=ustblue, fill=gray!6, rounded corners=2pt,
                minimum width=2.6cm, minimum height=0.7cm, align=center, font=\small},
    decision/.style={diamond, draw=ustgold_1, fill=ustgold_1!10, aspect=2.5,
                     align=center, font=\small, inner sep=1pt},
    arrow/.style={-{Stealth[length=5pt]}, thick, ustblue!80},
]
\node[box] (codegen) {Codegen OK\\answer valid};

\node[box, above right=0.25cm and 1.0cm of guard, fill=green!10] (fast) {Fast return\\(0 extra LLM)};
\node[box, below right=0.25cm and 1.0cm of guard] (plan) {plan(force=True)};
\node[decision, right=of plan] (ok) {Plan\\OK?};
\node[box, below right=0.2cm and 0.5cm of ok] (fallback) {Fallback plan\\from guard risks};
\node[box, above right=0.2cm and 0.5cm of ok] (judge) {Judge + Repair\\(max 3 rounds)};
\draw[arrow] (codegen) -- (guard);
\draw[arrow] (guard) -- node[above, font=\scriptsize]{No} (fast);
\draw[arrow] (guard) -- node[below, font=\scriptsize]{Yes} (plan);
\draw[arrow] (plan) -- (ok);
\draw[arrow] (ok) -- node[above, font=\scriptsize]{Yes} (judge);
\draw[arrow] (ok) -- node[below, font=\scriptsize]{No} (fallback);
\draw[arrow] (fallback) -| (judge);
\end{tikzpicture}%
}
\end{center}

\column{0.43\textwidth}
\textbf{Analyst-exception recovery (new):}\\
If analyst raises during plan, we used to fall through to a silently-skipped judge. Now:
\begin{itemize}
    \item \texttt{stage==analyst\_exception} $\to$ retry once with \texttt{force=True}
    \item Still fails $\to$ build low-confidence synthetic plan; \alert{judge must run}
\end{itemize}

\vspace{0.3em}
\textbf{Observability --- \texttt{semantic\_consistency\_audit}:}\\
Top-level trace key with \texttt{final\_gate}:
\begin{itemize}
    \scriptsize
    \item \texttt{plan+judge} --- strong path
    \item \texttt{escalated+judge} --- recovered
    \item \texttt{cheap\_guard\_only} --- investigate
    \item \texttt{bypassed} --- no semantic check ran
\end{itemize}
\end{columns}

\vspace{0.3em}
\begin{exampleblock}{Effect}
Simple tasks: \textbf{1 LLM call total} (codegen only). Hard tasks where analyst fails: guaranteed judge engagement instead of silent skip.
\end{exampleblock}

\end{frame}

\section{Supporting Components}

%---------------------------------------------------------
\begin{frame}
\frametitle{RAG Pipeline \& Multi-Agent Orchestrator}

\begin{columns}[T]
\column{0.55\textwidth}
\textbf{RAG (long\_docs route):}
\begin{itemize}
    \item Heading-aware chunking (1200 chars, overlap 150)
    \item \textbf{Query expansion}: paraphrases + HyDE
    \item \textbf{Hybrid retrieval}: BM25 $\parallel$ Embedding
    \item \textbf{RRF fusion} (rank-only, no score calibration)
    \item Optional cross-encoder reranker
    \item Embedding cache: \texttt{SHA256(model, text)}
\end{itemize}

\begin{frame}
\frametitle{Self-Consistency, Budget \& Demo}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Self-consistency --- column-signature voting:}
\begin{itemize}
    \item N passes at T=0.7 with different seeds
    \item \texttt{target\_count} = mode of column counts
    \item Top-\texttt{target\_count} signatures by frequency = winning set
    \item Pick sample with max coverage, min extras
    \item \alert{Directly targets} $\text{recall} - \lambda \cdot \text{extra/pred}$
\end{itemize}

\vspace{0.3em}
\textbf{Budget controller (per task):}\\
\scriptsize\texttt{max\_llm\_calls, max\_tool\_calls, max\_seconds, max\_local\_repairs, max\_reasoner\_repairs, max\_multiagent\_fallbacks}\\
{\footnotesize Limits = $\min$(YAML, TaskCompiler budget\_level)}

\column{0.48\textwidth}
\textbf{Streamlit demo:}
\begin{itemize}
    \item Real-time subprocess log (ProgressLogger)
    \item \textbf{Reasoning timeline}: schema $\to$ route $\to$ codegen $\to$ guard $\to$ judge
    \item Final answer table + generated Python
    \item Full \texttt{trace.json} (collapsible)
\end{itemize}

\vspace{0.3em}
\begin{lstlisting}[language=bash]
uv run streamlit run \
  scripts/demo_app.py
\end{lstlisting}

\vspace{0.2em}
Auditable, explainable, reproducible.
\end{columns}

\section{Conclusion}

\begin{frame}
\frametitle{Summary \& Future Work}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{What we built:}
\begin{itemize}
    \item Tool-first Data Agent pipeline
    \item Deterministic task profiling (zero LLM)
    \item Shape-based routing + failure-type cascade
    \item 10+ local repair rules (zero LLM)
    \item \alert{Cheap Semantic Guard} --- non-LLM risk gate
    \item Analyst-exception recovery + audit trail
    \item Production-grade RAG (BM25+Emb+RRF)
    \item Multi-agent DAG with bounded replan
    \item Column-signature self-consistency
\end{itemize}

\column{0.48\textwidth}
\textbf{Observed performance (public):}\\
Easy: $\sim$70--80 \quad Medium/Hard: $\sim$60

\vspace{0.4em}
\textbf{Future work:}
\begin{itemize}
    \item Vision executor for image tasks
    \item Guard-weight auto-calibration from trace FP/FN
    \item Per-value majority voting within columns
    \item Smarter record-text extraction for long prose
    \item Submit to hidden test after tuning
\end{itemize}
\end{columns}
