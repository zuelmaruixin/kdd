\section{Problem \& Architecture}

%---------------------------------------------------------
\begin{frame}
\frametitle{Task \& Scoring Rule}

\begin{columns}[T]
\column{0.45\textwidth}
\textbf{Input:} \texttt{task.json} + heterogeneous context\\
{\small (CSV, SQLite, JSON, Markdown rules, record-text prose, images)}

\vspace{0.3em}
\textbf{Output:} \texttt{prediction.csv} scored by\\
\textbf{column-content signature matching}

\vspace{0.3em}
\begin{itemize}
    \item Column names \textbf{ignored}; row order \textbf{ignored}
    \item Multi-column match via bipartite augmenting paths
    \item Normalize: numeric $\to 10^{-2}$, string \texttt{strip+lower}
\end{itemize}

\column{0.52\textwidth}
\vspace{0.5em}
\begin{equation*}
\text{score} = \max\!\Big(0,\; \text{recall} - \lambda \cdot \tfrac{\text{extra\_cols}}{\text{pred\_cols}}\Big), \;\; \lambda = 0.5
\end{equation*}

\vspace{0.3em}
\begin{alertblock}{Design implications}
\begin{itemize}
    \item \textbf{Fewer columns} --- every extra wrong col is penalized
    \item \textbf{Preserve granularity} --- \texttt{first\_name}+\texttt{last\_name} beats merged \texttt{name}
    \item \textbf{Match dtypes} --- \texttt{163109.0} $\neq$ \texttt{163109}
\end{itemize}
\end{alertblock}
\end{columns}

\end{frame}
%---------------------------------------------------------

\begin{frame}
\frametitle{Why Layered? --- Four-Layer Architecture}

\textbf{Vanilla ReAct breaks down:} easy tasks waste tokens on multi-turn tool calls; hard tasks drift in intermediate steps; no structural guarantee on output shape; no cheap way to verify semantic correctness.

\vspace{0.4em}
\begin{center}

\node[box, fill=ustgold_1!20, right=of router] (operator) {3. Operator Pipeline\\{\scriptsize codegen + repair}};

\begin{block}{Design principle}
\emph{Simple tasks pay minimal LLM cost; only suspicious answers escalate to expensive semantic verification.}
\end{block}

\section{Pipeline \& Routing}

%---------------------------------------------------------
\begin{frame}
\frametitle{TaskCompiler \& Router --- Dispatch by Shape, Not Difficulty}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{TaskCompiler (zero LLM):}
\begin{itemize}
    \item Per-file scan $\to$ \texttt{SourceCapability}\\{\scriptsize headers, dtypes, cardinality, samples}
    \item Task-level aggregation $\to$ \texttt{ExecutionProfile}
    \begin{itemize}
        \scriptsize
        \item \texttt{context\_size}, \texttt{source\_shape}
        \item \texttt{verifiability}, \texttt{operation\_complexity}
        \item \texttt{semantic\_rule\_required}
    \end{itemize}
    \item \alert{Single source of truth} --- reused by prompt, static checker, guard, judge
\end{itemize}

\column{0.48\textwidth}
\textbf{Router rules (hard, deterministic):}
\begin{itemize}
    \item \texttt{weak} / \texttt{image} $\to$ multi\_agent
    \item \texttt{long\_docs} $\to$ RAG-enabled operator
    \item \texttt{semantic\_rule\_required} $\to$ tool\_first\_mixed
    \item \texttt{small+programmatic+low} $\to$ easy
\end{itemize}

\vspace{0.3em}
\textbf{Cascade by \emph{failure type}, not difficulty:}\\
{\scriptsize zero\_rows $\to$ \alert{no upgrade} (column wrong, not route)\\
retrieval\_empty $\to$ RAG route\\
semantic\_consistency\_failed $\to$ tool\_first\_mixed\\
syntax/static/exec error $\to$ sibling operator}
\end{columns}

\vspace{0.3em}
\begin{exampleblock}{Cross-model verify (optional)}
Primary + verifiers run in parallel; keep columns whose signatures agree $\geq$ \texttt{min\_agreement} --- directly targets the $-\lambda \cdot$ extra penalty.
\end{exampleblock}

\end{frame}

\vspace{0.2em}
\begin{block}{Shared evidence --- \texttt{debug\_steps}}
Every generated program emits \texttt{schema\_inspection}, \texttt{used\_columns}, \texttt{filter\_conditions}, \texttt{join\_keys}, \texttt{intermediate\_counts}. Read by \alert{3 consumers}: cheap guard, preflight, LLM judge --- single audit trail, no drift.
\end{block}

\end{frame}

%---------------------------------------------------------
\begin{frame}
\frametitle{Schema Grounding + Static Checker + Local Repair}

\begin{columns}[T]
\column{0.32\textwidth}
\textbf{Schema Grounding}\\
{\scriptsize\textit{Tabular-source prompt hint}}
\begin{itemize}
    \item Levenshtein + value match + cardinality on real columns TOP-K to Prompt
    \item \textbf{Per source-kind policy:}
        \item \textbf{CSV / SQLite} --- full scoring
        \item \textbf{JSON} --- downgrade: inject sample + record\_fields into prompt (no Levenshtein)
        \item \textbf{text / image} --- skip
\end{itemize}

\column{0.32\textwidth}
\textbf{Static Checker}\\
{\scriptsize\textit{Pre-execution gate}}
\begin{itemize}
    \item Python AST + SQL scan
    \item \texttt{no\_such\_column}, \texttt{bad\_join\_key}, \texttt{\_x/\_y} suffixes
    \item Runs \emph{before} execution
    \item Blocks predictable failures
\end{itemize}

\column{0.32\textwidth}
\textbf{Local Repair (0 LLM)}\\
{\scriptsize\textit{Deterministic patcher}}
\begin{itemize}
    \item 10+ deterministic patches
    \item Fuzzy column-name fix
    \item Cast merge keys to string
    \item Append missing \texttt{answer}
    \item Zero-row filter normalize
\end{itemize}
\end{columns}

\vspace{0.4em}
\begin{alertblock}{Source-kind-agnostic defense (real safety net)}
Grounding is a prompt optimization, not a universal guard. The \alert{actual per-source defense} lives downstream and works on \emph{every} source kind: \textbf{schema\_inspection} (codegen must dump \texttt{list(df.columns)} after load) $\to$ \textbf{static checker} $\to$ \textbf{cheap guard} (reads trace, not source) $\to$ \textbf{LLM judge}. All four read the same \texttt{SourceCapability} scan --- one audit trail, no drift.
\end{alertblock}

\end{frame}

\section{Risk Gating (Core Innovation)}

%---------------------------------------------------------
%---------------------------------------------------------
\begin{frame}
\frametitle{Cheap Semantic Guard --- Non-LLM Risk Gate}

\textbf{Position:} codegen succeeded, answer structurally valid, $\to$ trust this answer?

\vspace{-0.2em}
\begin{equation*}
\text{risk\_score} = \min\!\big(1.0,\textstyle\sum \text{unique\_weights}\big), \quad
\text{escalate} = \text{any error} \,\lor\, \text{score} \geq 0.5
\end{equation*}

\vspace{-0.3em}
\begin{center}
\scriptsize
\begin{tabular}{llcl}
\toprule
\textbf{Category} & \textbf{Risk Code} & \textbf{Wt} & \textbf{Meaning} \\
\midrule
Execution & \texttt{execution\_failed / empty\_answer\_rows} & 1.0 / 0.75 & No/zero answer rows \\
Repair & \texttt{suspicious\_fallback:schema\_retry} & 0.8 & Needed schema retry \\
Trace & \texttt{trace\_missing:schema\_inspection} & 0.8 & No real-column check \\
Trace & \texttt{zero\_intermediate\_count} & 0.8 & Mid-pipeline drops to 0 \\
\textbf{Coverage} & \texttt{\textbf{field\_coverage\_mismatch}} & \textbf{0.75} & \textbf{Grounding candidate unused} \\
Formula & \texttt{formula\_without\_trigger} & 0.75 & Unasked formula used \\
\bottomrule
\end{tabular}
\end{center}

\vspace{-0.2em}
\begin{alertblock}{Killer example}
\scriptsize
Q: \emph{``patients with severe degree of thrombosis''}. Code uses \texttt{disease=="thrombosis"} but grounding found \texttt{degree\_of\_thrombosis} $\to$ \texttt{field\_coverage\_mismatch} (0.75) $\to$ \alert{escalate to judge} instead of shipping wrong answer.
\end{alertblock}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------

\begin{frame}[t]
\frametitle{Lazy Escalation + Analyst-Exception Recovery}

\vspace{-0.65em}
\footnotesize

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\textbf{1. Post-codegen gate}

\vspace{0.1em}
After codegen succeeds and the answer is structurally valid, the \textbf{Cheap Guard} scores 10+ deterministic rules (non-LLM). If \texttt{score < 0.5} \alert{$\to$ fast return}, zero extra LLM cost. Otherwise escalate.

\vspace{0.35em}

\textbf{2. Escalation flow (LLM path)}

\vspace{0.1em}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \textbf{Analyst} (1 LLM call) --- emits semantic plan
    \item \textbf{Judge} --- compares plan vs code
    \begin{itemize}
        \scriptsize
        \setlength\itemsep{0.03em}
        \item \texttt{pass} $\to$ done
        \item \texttt{fail} / \texttt{low\_confidence} $\to$ \textbf{Semantic Repair} rewrites code
    \end{itemize}
    \item Judge--Repair loops at most \textbf{3 rounds}
\end{itemize}

\vspace{0.25em}

\textbf{3. Analyst-exception recovery}
\texttt{stage=analyst\_exception} → retry once with \texttt{force=True}; still fails → low-confidence synthetic plan → \alert{judge must run}.

\column{0.49\textwidth}

\textbf{4. Observability --- \texttt{final\_gate}}

\vspace{0.05em}
Top-level field in \texttt{trace.json}:
\begin{itemize}
    \scriptsize
    \setlength\itemsep{0.05em}
    \item \texttt{plan+judge} --- strong path
    \item \texttt{escalated+judge} --- recovered (retry / fallback / lazy escalation)
    \item \texttt{cheap\_guard\_only} --- guard fired but judge did not run (investigate)
    \item \texttt{bypassed} --- no semantic check
\end{itemize}

\vspace{0.3em}

\textbf{5. Cost}

\vspace{0.05em}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item Simple task: \textbf{1 LLM call} (codegen only)
    \item Rule-based task: 3--5 calls (analyst + codegen + judge $\pm$ repair)
    \item Analyst exception: +1 retry +1 judge (worst case)
\end{itemize}
\vspace{0.1em}
\alert{\textbf{Core guarantee}}

\scriptsize
No suspicious answer ships unchecked: if guard escalates, judge \emph{will} run --- even if analyst fails.

\end{columns}
\end{frame}

\begin{frame}[t]
\frametitle{Multi-Agent Mode --- Roles \& Permissions}

\vspace{-0.7em}
\scriptsize

\begin{columns}[T,onlytextwidth]

% ================= LEFT COLUMN =================
\column{0.49\textwidth}

\textbf{1. Planner}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \textbf{Job:} decompose the question into a DAG of sub-tasks
    \item \textbf{Output:} JSON plan with dependencies
    \item \textbf{Tools:} none; pure reasoning only
    \item \textbf{Constraint:} cannot call \texttt{answer} or \texttt{report}
\end{itemize}

\vspace{0.25em}

\textbf{2. Specialists} (5 kinds, parallel execution)

\vspace{0.1em}
\tiny
\begin{tabular}{@{}p{0.19\linewidth}p{0.75\linewidth}@{}}
\toprule
\textbf{Kind} & \textbf{Allowed tools} \\
\midrule
Schema &
\texttt{list\_context, read\_csv, read\_json, read\_doc, inspect\_sqlite\_schema} \\

SQL &
\texttt{list\_context, read\_doc, inspect\_sqlite\_schema, execute\_context\_sql} \\

Python &
\texttt{list\_context, read\_csv/json/doc, inspect\_sqlite, execute\_sql, execute\_python} \\

Document &
\texttt{list\_context, read\_doc, read\_json, read\_csv} \\

Generic &
all tools; full ReAct \\
\bottomrule
\end{tabular}

\vspace{0.25em}
\scriptsize
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item Terminal tool = \texttt{report}, not \texttt{answer}
    \item \texttt{read\_doc} capped at 2 calls per specialist
    \item Max 12 ReAct steps per specialist
\end{itemize}

% ================= RIGHT COLUMN =================
\column{0.49\textwidth}

\textbf{3. Synthesizer}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \textbf{Job:} merge specialist findings into one final answer
    \item \textbf{Input:} all \texttt{report} payloads
    \item Only role allowed to call \texttt{answer}
    \item Max 6 ReAct steps
    \item Synthesis failure $\to$ one replan allowed
\end{itemize}

\vspace{0.25em}

\textbf{4. Orchestrator}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item Dispatches sub-tasks via ThreadPool
    \item Same-layer specialists run \textbf{in parallel}
    \item Enforces DAG dependencies
    \item Layer $N$ waits for layer $N-1$
    \item Iterative refinement: 1 replan max
    \item Budget cap: \texttt{max\_multiagent\_fallbacks = 1}
\end{itemize}

\vspace{0.25em}

\begin{block}{Key design: permission isolation}
\tiny
Specialists cannot submit the final answer. They only \texttt{report} findings. 
The Synthesizer must cross-check all reports before committing the final \texttt{answer}.
\end{block}

\end{columns}

\begin{frame}[fragile]
\frametitle{Self-Consistency, Budget \& Demo}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Self-consistency --- column-signature voting:}
\begin{itemize}
    \item N passes at T=0.7 with different seeds
    \item \texttt{target\_count} = mode of column counts
    \item Top-\texttt{target\_count} signatures by frequency = winning set
    \item Pick sample with max coverage, min extras
    \item \alert{Directly targets} $\text{recall} - \lambda \cdot \text{extra/pred}$
\end{itemize}

\vspace{0.5em}
\textbf{Budget controller (per task):}
\begin{itemize}
    \scriptsize
    \item LLM / tool / time caps
    \item Local \& reasoner repair caps
    \item Multi-agent fallback cap
    \item Limits $=\min$(YAML, TaskCompiler level)
\end{itemize}

\column{0.48\textwidth}
\textbf{Streamlit demo:}
\begin{itemize}
    \item Real-time subprocess log (ProgressLogger)
    \item \textbf{Reasoning timeline}: schema $\to$ route $\to$ codegen $\to$ guard $\to$ judge
    \item Final answer table + generated Python
    \item Full \texttt{trace.json} (collapsible)
\end{itemize}

\vspace{0.3em}
\begin{lstlisting}[language=bash]
uv run streamlit run \
  scripts/demo_app.py
\end{lstlisting}

\vspace{0.2em}
Auditable, explainable, reproducible.
