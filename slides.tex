\section{Problem \& Scoring Rule}

%---------------------------------------------------------
\begin{frame}
\frametitle{Task Definition}

\begin{block}{Input}
\texttt{task.json} (question + difficulty) + heterogeneous context files:\\
CSV, SQLite, JSON, Markdown rules, record-text prose, images
\end{block}

\vspace{0.4em}

\begin{block}{Output}
A \texttt{prediction.csv} table scored by \textbf{column-content signature matching}
\end{block}

\vspace{0.4em}

\begin{alertblock}{Key Characteristics}
\begin{itemize}
    \item Column \textbf{names ignored}; row \textbf{order ignored}
    \item Matching based on sorted, normalized value tuples per column
    \item Multi-column matching via \textbf{bipartite maximum matching} (augmenting path)
\end{itemize}
\end{alertblock}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}
\frametitle{Scoring Formula}

\begin{equation*}
\boxed{\text{score} = \max\!\Big(0,\;\; \text{recall} \;-\; \lambda \cdot \frac{\text{extra\_cols}}{\text{pred\_cols}}\Big), \quad \lambda = 0.5}
\end{equation*}

\vspace{0.6em}

\textbf{Implications for system design:}

\begin{enumerate}
    \item \textbf{Prefer fewer columns} --- every extra wrong column is penalized
    \item \textbf{Preserve granularity} --- \texttt{first\_name} + \texttt{last\_name} $>$ merged \texttt{name}
    \item \textbf{Normalize dtypes} --- \texttt{163109.0} $\neq$ \texttt{163109} in signature
\end{enumerate}

\vspace{0.6em}

\begin{exampleblock}{Normalization}
Numeric: round to $10^{-2}$; \quad String: \texttt{strip + lower}; \quad Empty/NaN $\to$ \texttt{\_\_EMPTY\_\_}
\end{exampleblock}

\end{frame}

\section{Architecture Overview}

%---------------------------------------------------------
\begin{frame}
\frametitle{Why Not a Single ReAct Agent?}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Problems with vanilla ReAct:}
\begin{itemize}
    \item Easy tasks: wastes tokens on multi-turn tool calls
    \item Hard tasks: drifts in intermediate steps
    \item No structural guarantee on output table shape
    \item Cannot cheaply verify semantic correctness
\end{itemize}

\column{0.48\textwidth}
\textbf{Our approach:}
\begin{itemize}
    \item \textbf{Deterministic profiling} $\to$ classify task shape upfront
    \item \textbf{One-shot codegen} $\to$ pandas/SQL program
    \item \textbf{Layered risk gating} $\to$ cheap guard before LLM judge
    \item \textbf{DAG planning} $\to$ only for truly complex tasks
\end{itemize}
\end{columns}

\end{frame}

\section{TaskCompiler \& ExecutionProfile}

%---------------------------------------------------------
\begin{frame}
\frametitle{TaskCompiler --- Deterministic Task Profiling (Zero LLM)}

\begin{columns}[T]
\column{0.52\textwidth}
\textbf{Per-file scan $\to$ \texttt{SourceCapability}:}
\begin{itemize}
    \item CSV/TSV: headers, row count, top-5 values, dtype vote, cardinality
    \item SQLite: \texttt{PRAGMA table\_info}, per-column samples
    \item JSON: top-level keys, record fields, per-field samples
    \item Record-text: regex pre-extraction (patient ID, dates, indicators)
\end{itemize}

\column{0.44\textwidth}
\textbf{Task-level aggregation:}
\begin{itemize}
    \item \texttt{task\_type}: table\_computation, mixed\_context, document\_qa, \ldots
    \item \texttt{operations}: filter, join, aggregate, compute, \ldots
    \item \texttt{foreign\_key\_candidates}: Jaccard overlap of value sets
    \item \texttt{ambiguity\_flags}: semantic rule, large doc, record text, \ldots
    \item \texttt{budget\_level} $\to$ max LLM / tool calls
\end{itemize}
\end{columns}

\vspace{0.4em}
\begin{exampleblock}{Key Property}
One scan, reused everywhere: prompt, static checker, guard, judge --- \textbf{single source of truth}.
\end{exampleblock}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}
\frametitle{ExecutionProfile --- 5-Dimensional Task Shape}

\begin{center}
\small
\begin{tabular}{lll}
\toprule
\textbf{Dimension} & \textbf{Values} & \textbf{Used By} \\
\midrule
\texttt{context\_size} & small / medium / large & Router, budget \\
\texttt{source\_shape} & single\_table / multi\_table / long\_docs / image & Router, RAG gate \\
\texttt{verifiability} & programmatic / evidence\_based / weak & Route selection \\
\texttt{operation\_complexity} & direct\_lookup / single\_filter / \ldots / multi\_hop & Fast-path skip \\
\texttt{semantic\_rule\_required} & true / false & Rule resolution \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.5em}

\textbf{Derived decisions:}
\begin{itemize}
    \item \texttt{recommended\_strategy}: operator / rag\_extract / multi\_agent / \ldots
    \item \texttt{blocked\_strategies}: e.g.\ small+programmatic task \textbf{cannot} jump to extreme
    \item \texttt{direct\_candidate\_role}: submit\_after\_verify / proposal\_only / disabled
\end{itemize}

\end{frame}

\section{Router --- Shape-Based Dispatch}

%---------------------------------------------------------
\begin{frame}
\frametitle{Router: Dispatch by Task Shape, Not Difficulty Label}

\begin{center}
\small
\begin{tabular}{llp{6.5cm}}
\toprule
\textbf{Route} & \textbf{Execution Kind} & \textbf{When Selected} \\
\midrule
easy & operator\_executor / tablellm\_direct & Light table task, small context, low complexity \\
medium & operator\_executor + RAG & Mixed-source, document context \\
hard & operator\_executor & Semantic rule required, multi-table \\
extreme & multi\_agent (DAG) & Weak verifiability, multi-hop planning \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.3em}

\textbf{\texttt{\_route\_for\_execution\_profile} hard rules:}
\begin{itemize}
    \item \texttt{verifiability=weak} or \texttt{image} $\to$ multi\_agent
    \item \texttt{long\_docs} $\to$ RAG-enabled operator (mandatory)
    \item \texttt{semantic\_rule\_required} $\to$ tool\_first\_mixed / hard
    \item \texttt{small + programmatic + low\_complexity} $\to$ easy (never extreme)
\end{itemize}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}
\frametitle{Cascade Fallback --- By Failure Type, Not Difficulty}

\begin{alertblock}{Key Insight}
Upgrading difficulty does not fix the \emph{type} of failure. We cascade by \textbf{failure category}.
\end{alertblock}

\vspace{0.3em}

\begin{center}
\small
\begin{tabular}{lp{7cm}}
\toprule
\textbf{Failure Type} & \textbf{Cascade Action} \\
\midrule
\texttt{zero\_rows} & \textbf{Do NOT upgrade} --- column selection is wrong, not route \\
\texttt{unsupported\_file / budget\_exhausted} & $\to$ fallback\_multi\_agent \\
\texttt{retrieval\_empty / doc\_context\_miss} & $\to$ RAG-enabled route \\
\texttt{semantic\_consistency\_failed} & $\to$ tool\_first\_mixed \\
\texttt{syntax / static / exec error} & $\to$ another operator route (block extreme for small tasks) \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.3em}

\begin{block}{Cross-Model Verification (optional)}
Run primary + verifier models in parallel $\to$ keep columns whose signatures agree ($\geq$ \texttt{min\_agreement}) $\to$ directly targets $-\lambda \cdot \text{extra}$ penalty.
\end{block}

\end{frame}

\begin{frame}
\frametitle{Schema Grounding + Static Checker + Local Repair}

\begin{columns}[T]
\column{0.32\textwidth}
\textbf{Schema Grounding}
\begin{itemize}
    \item Extract question concepts
    \item Match to real columns via Levenshtein + value match + cardinality
    \item Top-3 candidates with dtype/samples
    \item Fights \alert{column-name hallucination}
\end{itemize}

\column{0.32\textwidth}
\textbf{Static Checker}
\begin{itemize}
    \item Python AST + SQL literal scan
    \item Checks: no\_such\_file, no\_such\_table, no\_such\_column, bad\_join\_key
    \item Tracks DataFrame lineage (\texttt{\_x/\_y} suffixes)
    \item Runs \emph{before} execution
\end{itemize}

\column{0.32\textwidth}
\textbf{Local Repair (0 LLM)}
\begin{itemize}
    \item 10+ deterministic patches
    \item Fuzzy column name fix
    \item JSON records loader rewrite
    \item Cast merge keys to string
    \item Append missing \texttt{answer} assignment
    \item Zero-row filter normalize
\end{itemize}
\end{columns}

\vspace{0.4em}
\begin{block}{Shared Truth}
Prompt, static checker, and repair all read the \textbf{same} \texttt{SourceCapability} --- no drift between ``what we told the LLM'' and ``what we verify against''.
\end{block}

\end{frame}

\section{Cheap Semantic Guard (Core Innovation)}

%---------------------------------------------------------
\begin{frame}
\frametitle{Cheap Semantic Guard --- Non-LLM Risk Gate}

\begin{block}{Position in Pipeline}
After codegen succeeds + answer is structurally valid + \textbf{no LLM analyst ran yet} (fast path) $\to$ should we trust this answer or escalate?
\end{block}

\vspace{0.3em}

\textbf{Decision rule (purely deterministic, zero LLM cost):}
\begin{equation*}
\text{risk\_score} = \min\!\big(1.0,\;\textstyle\sum \text{unique\_risk\_weights}\big)
\end{equation*}
\vspace{-0.3em}
\begin{equation*}
\text{should\_escalate} = \text{any}(\text{severity}=\text{error}) \;\lor\; \text{risk\_score} \geq 0.5
\end{equation*}

\vspace{0.3em}
\textbf{Four evidence categories (no model calls):}
\begin{enumerate}
    \item Execution / structural status
    \item Repair history (schema-retry, local-repair log)
    \item Trace completeness (schema\_inspection, used\_columns, filters, joins)
    \item Schema grounding coverage (risk concepts $\leftrightarrow$ used fields)
\end{enumerate}

\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}
\frametitle{Risk Rules \& Weights (Selected)}

\begin{center}
\scriptsize
\begin{tabular}{llcl}
\toprule
\textbf{Category} & \textbf{Risk Code} & \textbf{Wt} & \textbf{Meaning} \\
\midrule
Execution & \texttt{execution\_failed} & 1.0 & No answer produced \\
Execution & \texttt{empty\_answer\_rows} & 0.75 & 0 rows \\
Repair & \texttt{suspicious\_fallback:schema\_retry} & 0.8 & Needed schema retry to succeed \\
Repair & \texttt{suspicious\_fallback:local\_repair} & 0.35 & Needed local patch \\
Trace & \texttt{trace\_missing:schema\_inspection} & 0.8 & No real-column check in trace \\
Trace & \texttt{trace\_unknown\_used\_columns} & 0.8 & Claims columns not in schema \\
Trace & \texttt{zero\_intermediate\_count} & 0.8 & Mid-pipeline drops to 0 rows \\
Grounding & \texttt{grounding\_low\_confidence} & 0.7 & Risk concept $<$ 0.68 match \\
\textbf{Coverage} & \texttt{\textbf{field\_coverage\_mismatch}} & \textbf{0.75} & \textbf{Risk concept has candidate but code didn't use it} \\
Formula & \texttt{formula\_without\_trigger} & 0.75 & Used formula question didn't ask for \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.3em}
\begin{alertblock}{Killer Example}
Q: ``patients with \textbf{severe degree of thrombosis}'' \\
Code uses \texttt{disease == "thrombosis"} but grounding found \texttt{degree\_of\_thrombosis} as candidate \\
$\to$ \texttt{field\_coverage\_mismatch} (0.75) $\to$ \textbf{escalate to semantic judge}
\end{alertblock}

\end{frame}

\section{Self-Consistency, Budget \& Demo}

%---------------------------------------------------------
\begin{frame}
\frametitle{Self-Consistency --- Column-Signature Voting}

\textbf{Why vote per-column, not per-table?} Because the official rubric scores per-column signatures.

\vspace{0.4em}

\begin{enumerate}
    \item Run \texttt{num\_samples} passes (temperature=0.7, different seeds)
    \item Per successful sample: compute column signatures (sorted normalized tuples)
    \item \texttt{target\_count} = mode of column counts across samples
    \item Vote: rank signatures by frequency $\to$ top \texttt{target\_count} = winning set
    \item Pick best sample (max coverage of winning set, min extra columns)
    \item Project answer to winning columns only
\end{enumerate}

\vspace{0.3em}
\begin{block}{Alignment with Scoring}
Directly targets $\text{score} = \text{recall} - \lambda \cdot \text{extra/pred}$: voting eliminates unstable extra columns while preserving stable correct ones.
\end{block}

\end{frame}
