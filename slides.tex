\section{问题与系统架构}

%---------------------------------------------------------
\begin{frame}
\frametitle{任务形式与评分函数}

\begin{columns}[T]
\column{0.45\textwidth}
\textbf{输入：} \texttt{task.json} + 异构上下文目录\\
{\small （CSV、SQLite、JSON、Markdown 规则文档、自然语言记录、图像）}

\vspace{0.3em}
\textbf{输出：} \texttt{prediction.csv}，由\\
\textbf{列内容签名匹配}评分

\vspace{0.3em}
\begin{itemize}
    \item 列名忽略；行顺序忽略
    \item 多列匹配走二部图最大匹配
    \item 归一化：数值 $\to 10^{-2}$，字符串 \texttt{strip+lower}
\end{itemize}

\column{0.52\textwidth}
\vspace{0.5em}
\begin{equation*}
\text{score} = \max\!\Big(0,\; \text{recall} - \lambda \cdot \tfrac{\text{extra\_cols}}{\text{pred\_cols}}\Big), \;\; \lambda = 0.5
\end{equation*}

\vspace{0.3em}
\begin{alertblock}{设计含义}
\begin{itemize}
    \item \textbf{列要少而准}：每多一列冗余都会扣分
    \item \textbf{保留粒度}：\texttt{first\_name}+\texttt{last\_name} 优于合并的 \texttt{name}
    \item \textbf{类型一致}：\texttt{163109.0} $\neq$ \texttt{163109}
\end{itemize}
\end{alertblock}
\end{columns}

\end{frame}

\begin{frame}
\frametitle{设计目标：失败模式与设计原则}

\footnotesize
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{六类可观察失败模式}
\begin{itemize}
    \setlength\itemsep{0.1em}
    \item \textbf{答案缺乏数据支撑}：依文件预览或先验直接作答 $\to$ 形状对、内容错。
    \item \textbf{字段语义绑定错误}：将概念词当列名/过滤值；目标实为字段值。
    \item \textbf{工具协议不稳定}：自由文本 JSON 易被转义/括号问题打断。
    \item \textbf{错误信息可利用性不足}：原始异常不足以指导自我纠正。
    \item \textbf{程序静态错误}：codegen 引用不存在的文件、表、列。
    \item \textbf{任务级边界控制}：验证、修复、路径切换可耗尽时间。
\end{itemize}

\column{0.5\textwidth}
\textbf{四条设计原则}
\begin{itemize}
    \setlength\itemsep{0.1em}
    \item \textbf{先扫描，再推理}：任何模型推理前先做无 LLM 的上下文画像。
    \item \textbf{工具行为显式化}：所有数据访问与计算经统一工具完成，每步动作写入 \texttt{trace.json}。
    \item \textbf{本地检查优先}：结构、静态、风险评估、可机械修复的错误尽量由确定性规则处理；必要时再升级到 LLM。
    \item \textbf{路径失败可复盘}：路由选择、失败原因、兜底路径与最终答案均结构化写入 trace。
\end{itemize}
\end{columns}

\vspace{0.4em}
\begin{block}{核心权衡}
\scriptsize
模型负责"规划与生成"，确定性模块负责"约束与验证"。这条边界使主路径在不牺牲可解释性的前提下保持稳定。
\end{block}

\end{frame}

\begin{frame}
\frametitle{相对 LangChain / LangGraph 的定位}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{LangChain 词汇}
\begin{itemize}
    \item Agent = 模型 + 工具 + 迭代循环
    \item 运行时图 = 模型节点、工具节点、提交/重试边
    \item 中间件控制上下文、工具访问与护栏
    \item LangSmith 风格 trace 用于调试
\end{itemize}

\column{0.48\textwidth}
\textbf{本系统对应实现}
\begin{itemize}
    \item \texttt{ModelAdapter} = 提供方抽象
    \item \texttt{ToolRegistry} = 强类型工具目录
    \item \texttt{ReActAgent} = 模型—工具循环
    \item \texttt{Orchestrator} = planner-specialist-synthesizer DAG
    \item \texttt{trace.json} = 记忆 + 审计
\end{itemize}
\end{columns}

\vspace{0.4em}
\begin{block}{定位}
本项目使用一个轻量的、LangGraph 风格的运行时，而非直接依赖 LangChain。该架构与 LangChain 适配器兼容：将 \texttt{ToolRegistry} 包装为 LangChain 工具，复用同一套路由器、验证器与答案校验器即可。
\end{block}

\end{frame}

\section{Agent 主循环与路由}

%---------------------------------------------------------
\begin{frame}
\frametitle{Agent 控制器：按任务画像分派}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{TaskCompiler（不调用 LLM）：}
\begin{itemize}
    \item 逐文件扫描 $\to$ \texttt{SourceCapability}\\{\scriptsize 列名、类型、基数、样本}
    \item 任务级聚合 $\to$ \texttt{ExecutionProfile}
    \begin{itemize}
        \scriptsize
        \item \texttt{context\_size}, \texttt{source\_shape}
        \item \texttt{verifiability}, \texttt{operation\_complexity}
        \item \texttt{semantic\_rule\_required}
    \end{itemize}
    \item \alert{统一事实快照} —— 由 prompt、静态检查、guard、judge 共享
\end{itemize}

\column{0.48\textwidth}
\textbf{路由规则（确定性）：}
\begin{itemize}
    \item 全部 7 种 \texttt{task\_type} 统一映射至 \alert{\texttt{react\_harness}} 作为首发路径
    \item 失败回退依据 \texttt{failure\_type} 选择：
    \begin{itemize}
        \scriptsize
        \item \texttt{retrieval\_empty} $\to$ RAG-enabled operator
        \item \texttt{semantic\_consistency\_failed} $\to$ \texttt{tool\_first\_mixed}
        \item \texttt{syntax / static / exec\_error} $\to$ 同类 operator
        \item \texttt{zero\_rows} $\to$ 不换路径，由 local repair 处理
    \end{itemize}
    \item 单题硬预算：仅 \texttt{max\_seconds=600}，LLM 与工具调用次数不设上限
\end{itemize}

\vspace{0.3em}
\textbf{Cascade 链：}\\
{\scriptsize \texttt{react\_harness} $\to$ \texttt{tool\_first\_mixed} $\to$ \texttt{extreme} $\to$ \texttt{fallback\_multi\_agent}\\至多额外尝试 1 次}
\end{columns}

\vspace{0.3em}
\begin{exampleblock}{Cross-model verification（可选）}
主路径与若干 verifier endpoint 并行求解，仅保留列签名一致数 $\geq$ \texttt{min\_agreement} 的列；直接抑制评分中的 $-\lambda \cdot$ extra 罚项。当前配置中关闭，以保留单题 10 分钟预算给主路径。
\end{exampleblock}

\end{frame}

%---------------------------------------------------------
\begin{frame}[t]
\frametitle{React Harness 主路径（一）：协议、上下文与知识引导}

\vspace{-0.3em}
\footnotesize
\begin{columns}[T,onlytextwidth]

\column{0.49\textwidth}

\textbf{1. Native tool calling 与文本回退}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \texttt{ToolSpec} 同源维护 prompt 描述与 tools-API JSON Schema，避免协议描述漂移
    \item 模型经 OpenAI function-call 协议调用工具
    \item 服务端返回纯文本时退化为 JSON 解析，含\textbf{三层容错}：fence 剥离 / 控制字符转义 / 未闭合括号补齐
    \item 模型一次返回多个 tool\_call 时只采纳第一个，且不记录被丢弃的，避免下一轮 orphan tool message 触发 API 拒绝
\end{itemize}

\vspace{0.3em}

\textbf{2. 工具结果缓存（per-task）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 缓存键 \texttt{= (action, sorted(action\_input))}，处理参数次序差异
    \item 仅缓存 9 类只读工具；\texttt{execute\_python} 与 \texttt{answer} 不缓存
    \item 缓存命中永不触发终止，避免重复触发 \texttt{answer} 终结
\end{itemize}

\vspace{0.3em}

\textbf{3. 切片式 observation}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \texttt{read\_csv} 支持 \texttt{columns\_only}/\texttt{offset} 分页
    \item \texttt{read\_doc/json} 按 \texttt{max\_chars} + \texttt{offset} 分页
    \item \texttt{grep\_doc} 仅返回命中行 + 上下文
\end{itemize}

\column{0.49\textwidth}

\textbf{4. 轻量 pre-loop planner（可选）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 单次 LLM 调用；仅基于文件清单（最多 25 项）；不读取内容
    \item Plan 注入为 user 消息前缀，每轮重构 messages 时持续可见
    \item Planner prompt 强约束第二步必须读 \texttt{knowledge / rule / glossary} 文档
    \item Plan 解析或调用失败时优雅降级；\texttt{skip\_for\_easy} 跳过简单题以省一次调用
\end{itemize}

\vspace{0.3em}

\textbf{5. System Prompt 强约束（schema-link first，then code）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \textbf{Knowledge file 不可议价}：\texttt{knowledge.md / rule / glossary} 为单一事实源；先验知识与之冲突时一律以文档为准；要求语义化应用规则而非照抄公式
    \item \textbf{样本行使用约束}：\texttt{read\_csv} 等预览行仅可推断 dtype / 格式 / join key，禁止据此输出最终答案
    \item \textbf{粒度保留约束}：与评分函数对齐，\texttt{first\_name} / \texttt{last\_name} 等不应合并
    \item 数值容差 + 字符串归一化提醒；附 spreadsheet / document 两类 few-shot 示例
\end{itemize}

\end{columns}
\end{frame}
%---------------------------------------------------------

%---------------------------------------------------------
\begin{frame}[t]
\frametitle{React Harness 主路径（二）：错误归类与 self-verify 闭环}

\vspace{-0.3em}
\footnotesize
\begin{columns}[T,onlytextwidth]

\column{0.49\textwidth}

\textbf{6. 确定性 retry hints（无 LLM）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 错误模式按工具分类：Python（10 类，覆盖 \texttt{KeyError}、\texttt{ParserError}、\texttt{python\_timeout} 等）、SQL（4 类）、Path（1 类）、Answer（3 类）
    \item 错误签名归一化：去 \texttt{line N} / 路径 / 引号，使同类错误能"碰头"
    \item \texttt{ErrorHistory} 跟踪重复次数：相同错误 $\geq 2$ 次时提示前置 \emph{"Stop retrying — switch tools or read the data first"}
    \item 即使工具抛异常未返回 result，仍跑一次 hint 归类（覆盖 \texttt{answer} 行宽错误等）
\end{itemize}

\vspace{0.3em}

\textbf{7. Native tool\_call orphan 修复}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 工具异常时仍把 action 名 + tool\_call\_id 记录入 step
    \item 下一轮 messages 重建时不会出现 orphan tool message，避免 API 因 \texttt{tool\_call\_id} 不匹配而 400
\end{itemize}

\column{0.49\textwidth}

\textbf{8. 强制 self-verification（draft → verify → final）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \texttt{verification\_rounds = 1} + \texttt{use\_answer\_guard = true}
    \item 第一次 \texttt{answer} 不立即终止，被改写为 \texttt{status = draft\_submitted} observation，附 \texttt{draft\_columns / draft\_row\_count / remaining\_rounds}
    \item \alert{Cheap Answer Guard}（下一帧详述）输出风险代码 + top\_risk
    \item \texttt{build\_verify\_instructions} 把每个风险代码翻译为具体行动指令；附 \emph{Most important: ...} 突出最高风险
\end{itemize}

\vspace{0.3em}

\textbf{9. 预算与未确认的 draft 兜底}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item self-verify 阶段触发 \texttt{BudgetExceeded} 时，commit 最近 draft 并标记 \texttt{budget\_exceeded\_during\_self\_verify}
    \item 循环结束模型未确认时，回退至最近 draft 并标记未完成 verify，避免有效结果被整体丢弃
    \item 模型 API 抛异常时单独记录 \texttt{\_\_model\_error\_\_} step 并终止当前循环
\end{itemize}

\vspace{0.3em}

\textbf{10. 可观测性}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 每步含 \texttt{stream\_label}（如 \texttt{react step N}、\texttt{react.planner}）
    \item Guard 单独发 \texttt{react\_answer\_guard} 事件
    \item \texttt{failure\_reason} 区分多种结局：超时 commit draft / 未完成 verify / 超步数未提交答案
\end{itemize}

\end{columns}
\end{frame}

\begin{frame}
\frametitle{React Cheap Answer Guard：7 类确定性风险}

\vspace{-0.2em}
\scriptsize

\section{风险闸门（核心创新）}

%---------------------------------------------------------
\begin{frame}
\frametitle{Cheap Semantic Guard：非 LLM 风险闸门}

\textbf{位置：}codegen 已经成功，answer 也通过结构校验 $\to$ 是否可以信任该答案？

\vspace{-0.2em}
\begin{equation*}
\text{risk\_score} = \min\!\big(1.0,\textstyle\sum \text{unique\_weights}\big), \quad
\text{escalate} = \text{任一 error 级风险} \,\lor\, \text{score} \geq 0.5
\end{equation*}

\vspace{-0.3em}
\begin{center}
\scriptsize
\begin{tabular}{llcl}
\toprule
\textbf{类别} & \textbf{风险代码} & \textbf{权重} & \textbf{含义} \\
\midrule
执行 & \texttt{execution\_failed / empty\_answer\_rows} & 1.0 / 0.75 & 答案缺失或零行 \\
修复 & \texttt{suspicious\_fallback:schema\_retry} & 0.8 & 依赖 schema retry \\
Trace & \texttt{trace\_missing:schema\_inspection} & 0.8 & 缺真实列检查 \\
Trace & \texttt{zero\_intermediate\_count} & 0.8 & 中间行数降为 0 \\
\textbf{覆盖} & \texttt{\textbf{field\_coverage\_mismatch}} & \textbf{0.75} & \textbf{Grounding 候选未被使用} \\
公式 & \texttt{formula\_without\_trigger} & 0.75 & 使用了问题未要求的派生公式 \\
\bottomrule
\end{tabular}
\end{center}

\vspace{-0.2em}
\begin{alertblock}{典型识别场景}
\scriptsize
问题询问"具有特定属性的实体"，但程序使用与属性名仅形似的字段做过滤；schema grounding 已给出更合适的候选字段，却未被代码使用 $\to$ \texttt{field\_coverage\_mismatch}（0.75）$\to$ \alert{升级到 judge}，避免直接提交错误答案。
\end{alertblock}

\end{frame}

\begin{frame}[t]
\frametitle{惰性升级 + Analyst 异常恢复}

\vspace{-0.65em}
\footnotesize

\begin{columns}[T,onlytextwidth]

\column{0.48\textwidth}

\textbf{1. codegen 后的闸门}

\vspace{0.1em}
codegen 成功且答案结构合法后，\textbf{Cheap Guard} 用 13 类确定性风险打分（无 LLM）。若 \texttt{score < 0.5} 则 \alert{快速通过}，零额外 LLM 成本；否则升级。

\vspace{0.35em}

\textbf{2. 升级后的 LLM 流程}

\vspace{0.1em}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \textbf{审题官（Analyst）}（1 次 LLM 调用）—— 输出语义计划
    \item \textbf{执行官（Judge）} —— 比较计划与代码
    \begin{itemize}
        \scriptsize
        \setlength\itemsep{0.03em}
        \item \texttt{pass} $\to$ 完成
        \item \texttt{fail / low\_confidence} $\to$ \textbf{语义修复} 重写代码
    \end{itemize}
    \item Judge--Repair 循环最多 \textbf{3 轮}
\end{itemize}

\vspace{0.25em}

\textbf{3. Analyst 异常恢复}\\
\texttt{stage=analyst\_exception} $\to$ 以 \texttt{force=True} 重试一次；仍失败则构造低置信度的 synthetic plan $\to$ \alert{judge 必须运行}。

\column{0.49\textwidth}

\textbf{4. 可观测性 —— \texttt{final\_gate}}

\vspace{0.05em}
\texttt{trace.json} 顶层字段：
\begin{itemize}
    \scriptsize
    \setlength\itemsep{0.05em}
    \item \texttt{plan+judge} —— 主路径
    \item \texttt{escalated+judge} —— 经过重试 / fallback / 惰性升级后恢复
    \item \texttt{cheap\_guard\_only} —— guard 触发但 judge 未运行（需调查）
    \item \texttt{bypassed} —— 未进入语义验证
\end{itemize}

\vspace{0.3em}

\textbf{5. 成本}

\vspace{0.05em}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 简单任务：\textbf{1 次 LLM 调用}（仅 codegen）
    \item 含规则的任务：3--5 次（analyst + codegen + judge $\pm$ repair）
    \item Analyst 异常：最坏额外 +1 重试 +1 judge
\end{itemize}
\vspace{0.1em}
\alert{\textbf{核心保证}}

\scriptsize
任何可疑答案都不会被直接提交：一旦 guard 决定升级，judge 必然会运行 —— 即便 analyst 失败。

\end{columns}
\end{frame}

\begin{frame}[t]
\frametitle{工具沙箱、知识工具与长文档检索}

\vspace{-0.3em}
\footnotesize
\begin{columns}[T]

\column{0.34\textwidth}
\textbf{1. 受控执行环境}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item \texttt{execute\_python} 在独立 \texttt{multiprocessing.Process} 中运行；cwd 固定为任务 \texttt{context/}
    \item stdout / stderr 经 Queue 回传；超过 \textbf{30 秒} \texttt{terminate()/kill()}
    \item \texttt{execute\_context\_sql} 仅允许只读语句
    \item 文件读取均切片：\texttt{read\_csv} 支持 \texttt{columns\_only}/\texttt{offset}；\texttt{read\_doc/json} 按 \texttt{max\_chars} 分页；\texttt{grep\_doc} 返回命中上下文
\end{itemize}

\column{0.32\textwidth}
\textbf{2. \texttt{consult\_knowledge}（可选）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 仅当 \texttt{helper\_model.enabled} 且当前 route 为 React 时注册
    \item 自动扫描 \texttt{knowledge / rule / glossary / definition / schema\_notes} 命名的文档
    \item 由更聚焦的 helper 模型解释规则；约束其只能基于给定文档；找不到答案必须返回 \texttt{not\_specified}
    \item 把"读规则"与"算数据"解耦：主路径仍由工具完成数据计算
\end{itemize}

\column{0.32\textwidth}
\textbf{3. RAG（cascade 才激活）}
\begin{itemize}
    \setlength\itemsep{0.05em}
    \item 标题感知切片（1200 字符，重叠 150）
    \item BM25 稀疏；可加 dense embedding，按 \textbf{RRF} 融合
    \item 查询扩展：改写 + HyDE 假设回答
    \item 可选 cross-encoder reranker（先取 \texttt{first\_stage\_top\_n}=30，再 rerank 至 top-K=6）
    \item 渐进增强：组件缺失时退化为 BM25
\end{itemize}
\end{columns}

\vspace{0.3em}
\begin{block}{React 主路径上的文档处理}
\scriptsize
React Harness 不进入 RAG 流水线；文档检索由模型通过 \texttt{read\_doc} / \texttt{grep\_doc} / \texttt{head\_doc} 自行完成。RAG 仅当 cascade 进入 \texttt{tool\_first\_mixed} 或 \texttt{extreme} 后激活。
\end{block}

\end{frame}

\begin{frame}[t]

\begin{columns}[T,onlytextwidth]

\begin{itemize}

    \item 同层 specialist 可\textbf{并行}
    \item 强制执行 DAG 依赖关系
    \item 第 $N$ 层等待第 $N-1$ 层完成
    \item 迭代 refinement：至多 1 次重新规划
    \item 预算上限：\texttt{max\_multiagent\_fallbacks = 1}
\end{itemize}

\begin{block}{关键设计：权限隔离}
\tiny
Specialist 不可直接提交答案，只能 \texttt{report} 中间证据。

    \setlength\itemsep{0.05em}
    \item \textbf{列投影不够保守}：评分函数对额外列设惩罚；中间调试列未被剔除。
    \item \textbf{字段语义绑定}：自然语言概念被映射到近似但错误的字段。
    \item \textbf{语义修复后的静态错误}：重写引入新的列名或语法问题。
    \item \textbf{单题超时}：复杂路径耗尽 600 秒上限。
\end{itemize}

\vspace{0.4em}
\begin{block}{执行轨迹观察}
\scriptsize
绝大多数任务由 \texttt{react\_harness} 直接完成；进入 cascade 的任务比例较低，与系统设计定位一致——主路径承担稳定性与协议约束，cascade 路径处理程序化与语义风险较高的失败样例。
\end{block}
